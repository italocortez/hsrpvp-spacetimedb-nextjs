    const rows: object[] = [];
    for (const c of raw) {
        if (!c.cost) continue;
        const csId = c.cost.cost_set_id ?? 0;
        const modes = Object.keys(c.cost).filter(k => k !== 'cost_set_id');
        for (const rawMode of modes) {
            const modeBlock = c.cost[rawMode] as RawModeEidolonBlock | undefined;
            if (!modeBlock || typeof modeBlock !== 'object') continue;

            const classicCosts = extractEidolonCost(modeBlock.classic);

            // D-09 rule: both sub-blocks absent → skip the mode entirely (no row).
            if (!classicCosts && !auctionBaseBid) continue;

            const gameMode = snakeToPascalMode(rawMode);

         if (!modeBlock || typeof modeBlock !== 'object') return false;
         const keys = Object.keys(modeBlock as object);
         return !keys.includes('classic') && !keys.includes('auction') &&

       function migrateCost(cost: Record<string, any>): { changed: boolean; cost: Record<string, any> } {
         let changed = false;
         const out: Record<string, any> = {};
         for (const [k, v] of Object.entries(cost)) {

           if (needsMigration(v)) { out[k] = { classic: v }; changed = true; }
