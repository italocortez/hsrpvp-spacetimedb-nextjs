**A. Create `spacetimedb/src/reducers/ban.ts` with two reducers (D-06, D-07, D-08):**

```typescript
import spacetimedb from '../schema';
import { t, SenderError } from 'spacetimedb/server';
import { getAuthenticatedUser } from '../helpers/ensurePermissions';
import { auditInsert } from '../helpers/auditColumns';
import { performUserDeletion } from '../helpers/userDeletionHelper';

/**
 * Admin-only: Ban a provider ID (Discord or Google).
 * Also soft-deletes the linked user if one exists (D-08 enforcement point 3).
 */
export const admin_ban_provider = spacetimedb.reducer({
    banTypeTag: t.string(),   // 'DiscordId' or 'GoogleId'
    providerId: t.string(),
    reason: t.string(),
}, (ctx, { banTypeTag, providerId, reason }) => {
    const { user } = getAuthenticatedUser(ctx);

    // Only Admin can ban (not Moderator)
    if (user.role.tag !== 'Admin') {
        throw new SenderError('Only Admins can ban provider IDs.');
    }

    // Validate banTypeTag
    if (banTypeTag !== 'DiscordId' && banTypeTag !== 'GoogleId') {
        throw new SenderError('banTypeTag must be DiscordId or GoogleId');
    }
    if (!providerId || providerId.length === 0) {
        throw new SenderError('providerId is required');
    }
    if (!reason || reason.length === 0) {
        throw new SenderError('reason is required');
    }

    // Check if already banned
    const existing = [...ctx.db.BanRecord.provider_id.filter(providerId)]
        .find((b: any) => b.banType.tag === banTypeTag);
    if (existing) {
        throw new SenderError(`Provider ID ${providerId} is already banned as ${banTypeTag}`);
    }

    // Insert BanRecord
    ctx.db.BanRecord.insert({
        id: 0, // autoInc
        banType: { tag: banTypeTag, value: {} } as any,
        providerId,
        reason,
        bannedByUserId: user.id,
        ...auditInsert(ctx, user.id),
    });

    // D-08 enforcement point 3: find linked user and soft-delete them
    // Search UserPrivate for a row with this providerId
    let linkedUserId: number | null = null;
    if (banTypeTag === 'DiscordId') {
        const matches = [...ctx.db.UserPrivate.discord_id.filter(providerId)];
        if (matches.length > 0) linkedUserId = matches[0].userId;
    } else {
        const matches = [...ctx.db.UserPrivate.google_id.filter(providerId)];
        if (matches.length > 0) linkedUserId = matches[0].userId;
    }

    if (linkedUserId !== null && linkedUserId !== user.id) {
        // Soft-delete the banned user — triggers auto-logout via useAuth.ts 4s delay
        const bannedUser = ctx.db.User.id.find(linkedUserId);
        if (bannedUser && !bannedUser.deletedAt) {
            performUserDeletion(ctx, linkedUserId, user.id);
        }
    }

    console.log(`[BAN] Admin #${user.id} banned ${banTypeTag} ${providerId}: ${reason}`);
});

/**
 * Admin-only: Unban a provider ID (D-07: admins lift bans by deleting the record).
 */
export const admin_unban_provider = spacetimedb.reducer({
    banRecordId: t.u32(),
}, (ctx, { banRecordId }) => {
    const { user } = getAuthenticatedUser(ctx);

    if (user.role.tag !== 'Admin') {
        throw new SenderError('Only Admins can unban provider IDs.');
    }

    const record = ctx.db.BanRecord.id.find(banRecordId);
    if (!record) {
        throw new SenderError(`BanRecord #${banRecordId} not found`);
    }

    ctx.db.BanRecord.id.delete(banRecordId);
    console.log(`[UNBAN] Admin #${user.id} removed ban #${banRecordId} (${record.banType.tag} ${record.providerId})`);
});
```
