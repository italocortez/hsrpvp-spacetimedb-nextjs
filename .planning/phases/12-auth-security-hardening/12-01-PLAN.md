```typescript
/**
 * Check if a provider ID is banned. Uses provider_id btree index for O(log n) lookup.
 * Returns the first matching BanRecord or null.
 */
export function checkProviderBan(ctx: any, providerId: string): any | null {
    const bans = [...ctx.db.BanRecord.provider_id.filter(providerId)];
    return bans.length > 0 ? bans[0] : null;
}

/**
 * Check if a user has any linked provider that is banned.
 * Looks up UserPrivate by userId, then checks each non-null providerId against BanRecord.
 * Returns the first matching BanRecord or null.
 */
export function checkUserBanned(ctx: any, userId: number): any | null {
    const priv = ctx.db.UserPrivate.userId.find(userId);
    if (!priv) return null;
    if (priv.discordId) {
        const ban = checkProviderBan(ctx, priv.discordId);
        if (ban) return ban;
    }
    if (priv.googleId) {
        const ban = checkProviderBan(ctx, priv.googleId);
        if (ban) return ban;
    }
    return null;
}
```

| UserPrivate table → server code | Private table only accessible via views and reducers; no direct client subscription |
| BanRecord table → server code | Private table only accessible via admin views and reducers |

| T-12-02 | I (Info Disclosure) | UserPrivate table | mitigate | Table created without `public:` key (defaults to private); no client subscription possible |
| T-12-03 | I (Info Disclosure) | BanRecord table | mitigate | Table created without `public:` key; ban reasons and provider IDs not exposed to clients |
