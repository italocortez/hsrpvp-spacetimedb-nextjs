**Step 1 — Tighten mergeForUpdate signature (L91-111):**

```ts
// BEFORE
function mergeForUpdate<T extends Record<string, any>>(
    existing: T,
    incoming: Record<string, any>,
    fields: (keyof T)[]
): T {
    const merged: T = { ...existing };
    for (const f of fields) {
        const key = f as string;
        if (!(key in incoming)) {
            throw new Error(`mergeForUpdate: missing key '${key}' — validateKeys contract broken`);
        }
        if (incoming[key] != null) {
            (merged as any)[key] = incoming[key];
        }
    }
    return merged;
}

// AFTER
function mergeForUpdate<T extends Record<string, any>, K extends keyof T & string>(
    existing: T,
    incoming: Partial<Record<K, unknown>>,
    fields: readonly K[]
): T {
    const merged: T = { ...existing };
    for (const f of fields) {
        if (!(f in incoming)) {
            throw new Error(`mergeForUpdate: missing key '${f}' — validateKeys contract broken`);
        }
        if (incoming[f] != null) {
            (merged as any)[f] = incoming[f];  // Internal cast stays — this is not the caller-side as any
        }
    }
    return merged;
}
```
Note: the internal `(merged as any)[f]` cast at the write site stays. That's the D-03b "internal helper" as-any excluded from phase scope.
