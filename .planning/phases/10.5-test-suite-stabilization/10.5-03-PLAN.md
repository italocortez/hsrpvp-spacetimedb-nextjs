<!-- Cleanup pattern for ACHIEVEMENT-creating tests -->
Achievement tracking + cleanup pattern:
```typescript
const createdAchievementIds: number[] = [];

// Inside each it() that creates an achievement:
await admin.call.createAchievement({ ... });
await admin.sync(1500);
const achievementId = [...admin.conn.db.Achievement.iter()]
  .sort((a, b) => b.id - a.id)[0].id;  // most recent
createdAchievementIds.push(achievementId);

// afterAll:
afterAll(async () => {
  for (const id of createdAchievementIds) {
    try { await admin.call.deleteAchievement({ achievementId: id }); } catch (_) {}
  }
  await admin?.disconnect();
});
```
