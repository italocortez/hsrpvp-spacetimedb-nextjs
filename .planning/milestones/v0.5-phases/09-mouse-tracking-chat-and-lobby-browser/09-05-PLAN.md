**Create spacetimedb/src/reducers/lobbyGc.ts (per D-25):**

Export the raw reducer function for the scheduled table reference, plus a helper to extract the cascade delete logic from close_lobby.

```typescript
import spacetimedb from '../schema';
import { SYSTEM_USER_ID } from '../helpers/auditColumns';

// Forward declaration for scheduled table reference
export const _runLobbyGcReducer = spacetimedb.reducer(
    { arg: /* LobbyGcJob.rowType — import from table */ },
    (ctx, { arg }) => {
        const now = ctx.timestamp;
        const THIRTY_MINUTES_MICROS = BigInt(30 * 60 * 1_000_000);

        // Iterate all lobbies
        for (const lobby of ctx.from.Lobby) {
            // Only clean Waiting and Finished stages per D-25
            if (lobby.stage.tag !== 'Waiting' && lobby.stage.tag !== 'Finished') continue;

            // Check idle time
            const idleTime = now.microsSinceUnixEpoch - lobby.lastActivityAt.microsSinceUnixEpoch;
            if (idleTime < THIRTY_MINUTES_MICROS) continue;

            // Hard delete cascade — same as close_lobby
            hardDeleteLobby(ctx, lobby.id);
        }

        // Schedule next GC run in 5 minutes
        const FIVE_MINUTES = { tag: 'Interval', value: BigInt(5 * 60 * 1_000_000) };
        ctx.db.LobbyGcJob.insert({
            scheduledId: BigInt(0),
            scheduledAt: FIVE_MINUTES as any,
        });
    }
);
```
