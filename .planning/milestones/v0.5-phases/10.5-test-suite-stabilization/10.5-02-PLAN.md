// test/shared/helpers/promoteUser.ts
import { DbConnection } from '@/src/module_bindings';

const DB = process.env.SPACETIMEDB_DB ?? 'hsrpvp-spacetimedb-nextjs-test1';

export async function promoteUser(username: string, roleTag: string): Promise<void> {
  const host = process.env.SPACETIMEDB_URI ?? 'wss://maincloud.spacetimedb.com';
  const token = process.env.SPACETIMEDB_SERVER_TOKEN!;
  if (!token) throw new Error('SPACETIMEDB_SERVER_TOKEN required for promoteUser');
  return new Promise((resolve, reject) => {
    const timeout = setTimeout(() => reject(new Error('Promote timeout')), 10000);
    DbConnection.builder()
      .withUri(host).withDatabaseName(DB).withToken(token)
      .onConnect((conn) => {
        conn.reducers.serverSetRole({ username, roleTag });
        setTimeout(() => { clearTimeout(timeout); resolve(); }, 1000);
      })
      .onConnectError((_ctx, err) => { clearTimeout(timeout); reject(err); })
      .build();
  });
}

// Convenience wrappers
export async function promoteToAdmin(username: string): Promise<void> {
  return promoteUser(username, 'Admin');
}

export async function promoteToRole(h: { userId: number; conn: { db: { User: { iter: () => Iterable<{ id: number; username: string }> } } } }, roleTag: string): Promise<void> {
  const user = [...h.conn.db.User.iter()].find(u => u.id === h.userId);
  if (!user) throw new Error(`User not found for userId ${h.userId}`);
  return promoteUser(user.username, roleTag);
}
